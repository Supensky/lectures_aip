// копирует строку src вместе с завершающим нулем в dest

char *strcat(char * restrict dest, const char * restrict src);
// Дописывает строку src в конец строки dest (включая '\0')

int strcmp(const char *s1, const char *s2);
// Сравнивает две строки лексикографически. Возвращает отрицательное, нуль или положительное значение

char *strchr(const char *s, int c);

char *strstr(const char *haystack, const char *needle);
// Ищет первое вхождение подстроки needle в строки haystack
